                          <FieldError errors={[err?.total]} />

              {despesasArray.fields.map((row, index) => {

                  <Field data-invalid={Boolean(err?.descricao)}><FieldLabel htmlFor={fieldId + "despesa" + index}>Descrição</FieldLabel><Input id={fieldId + "despesa" + index} aria-invalid={Boolean(err?.descricao)} placeholder="Ex.: Compra de gelo" {...form.register(`despesas.${index}.descricao`, { onChange: () => revalidarErro(`despesas.${index}.descricao`), validate: (v) => (v.trim().length >= 2 && v.trim().length <= 200) || "Descreva a despesa (2 a 200 caracteres)." })} /><FieldError errors={[err?.descricao]} /></Field>
